WITH RECURSIVE ancestors(id) AS (

SELECT max(id) FROM commits -- HEAD

UNION

SELECT p.parent_id FROM ancestors a JOIN commit_parents p ON p.commit_id = a.id

)

SELECT count(*) FROM ancestors;

SELECT cn.name,

COUNT(*) AS titles

FROM title AS t,

movie_companies AS mc,

company_name AS cn

WHERE t.id = mc.movie_id

AND mc.company_id = cn.id

GROUP BY cn.name

ORDER BY titles DESC

LIMIT 10;
